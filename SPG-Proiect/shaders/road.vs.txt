#version 330 core
layout (location = 0) in vec3 aPos;
layout (location = 1) in vec2 aTexCoords;

out vec2 TexCoords;

uniform mat4 model;
uniform mat4 view;
uniform mat4 projection;

void main()
{
    TexCoords = aTexCoords;
    
    vec3 elevatedPos = aPos;
    elevatedPos.y += 0.5; 
    
    gl_Position = projection * view * model * vec4(elevatedPos, 1.0);
}
