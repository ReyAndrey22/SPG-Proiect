#version 330 core
out vec4 FragColor;

in vec3 FragPos;
in vec3 Normal;
in vec2 TexCoords;

struct PointLight {
    vec3 position;
    vec3 color;
    float constant;
    float linear;
    float quadratic;
};

#define NUM_LIGHTS 6

uniform sampler2D texture_diffuse;
uniform vec3 viewPos;

uniform vec3 dirLightDir;
uniform vec3 dirLightColor;

uniform PointLight pointLights[NUM_LIGHTS];

vec3 CalcDirLight(vec3 normal, vec3 texColor)
{
    vec3 lightDir = normalize(-dirLightDir);
    float diff = max(dot(normal, lightDir), 0.0);
    return diff * dirLightColor * texColor;
}

vec3 CalcPointLight(PointLight light, vec3 normal, vec3 fragPos, vec3 texColor)
{
    vec3 lightDir = normalize(light.position - fragPos);
    float diff = max(dot(normal, lightDir), 0.0);
    vec3 diffuse = diff * light.color * texColor;
    

    float distance = length(light.position - fragPos);
    float attenuation = 1.0 / (light.constant + light.linear * distance + light.quadratic * (distance * distance));    
    
    return diffuse * attenuation;
}

void main()
{           
    vec3 norm = normalize(Normal);
    vec3 texColor = texture(texture_diffuse, TexCoords).rgb;
    

    vec3 ambient = vec3(0.15) * texColor;
    

    vec3 lightingResult = CalcDirLight(norm, texColor);
    

    for(int i = 0; i < NUM_LIGHTS; i++) {
        lightingResult += CalcPointLight(pointLights[i], norm, FragPos, texColor);
    }
        
    FragColor = vec4(ambient + lightingResult, 1.0);
}
