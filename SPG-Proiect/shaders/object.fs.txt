#version 330 core
in vec3 FragPos;
in vec3 Normal;
in vec2 TexCoords;

out vec4 FragColor;

uniform sampler2D texture_diffuse;

void main()
{
    vec3 norm = normalize(Normal);
    vec3 lightDir = normalize(vec3(1.0, 1.0, 1.0) - FragPos);
    
    float diff = max(dot(norm, lightDir), 0.0);
    vec3 diffuse = diff * texture(texture_diffuse, TexCoords).rgb;
    
    vec3 ambient = 0.3 * texture(texture_diffuse, TexCoords).rgb;
    
    FragColor = vec4(ambient + diffuse, 1.0);
}
